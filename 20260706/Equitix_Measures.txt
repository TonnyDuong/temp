Margin FY % = DIVIDE([Profit FY], [Total Revenue FY])

*Margin YTD % = DIVIDE([Profit YTD], [Total Revenue YTD])

Profit FY = [Total Revenue FY] - [Total Costs FY]

*Profit YTD = [Total Revenue YTD] - [Total Costs YTD]

*Revenue = SUM(fact_Revenue_Live[Amount]) * -1

Revenue Act In-Contract =
CALCULATE(
    -SUM(fact_Revenue_Live[Amount]),
    fact_Revenue_Live[Type] = "Actual",
    fact_Revenue_Live[Contract Type] = "In Contract")

Revenue Act Oo-Contract =
CALCULATE(
    -SUM(fact_Revenue_Live[Amount]),
    fact_Revenue_Live[Type] = "Actual",
    fact_Revenue_Live[Contract Type] = "Out of Contract",
    fact_Revenue_Live[Account Code] = "40013")

Revenue Actual = CALCULATE([Revenue], fact_Revenue_Live[Type] = "Actual")

Revenue For In-Contract =
CALCULATE(
    SUM(fact_Revenue_Live[Amount]),
    fact_Revenue_Live[Type] = "Forecast",
    fact_Revenue_Live[Contract Type] = "In Contract")

Revenue For Oo-Contract =
CALCULATE(
    SUM(fact_Revenue_Live[Amount]),
    fact_Revenue_Live[Type] = "Forecast",
    fact_Revenue_Live[Contract Type] = "Out of Contract")

Revenue Forecast Remaining =
[Revenue For In-Contract] + [Revenue For Oo-Contract]

Staff Cost Actual =
CALCULATE(
    SUM(fact_Cost_Live[Amount]),
    fact_Cost_Live[Type] = "Actual",
    fact_Cost_Live[Category] = "Staff")

Staff Cost Forecast =
CALCULATE(
    SUM(fact_Cost_Live[Amount]),
    fact_Cost_Live[Type] = "Forecast",
    fact_Cost_Live[Category] = "Staff")

Subcontractor Actual =
CALCULATE(
    SUM(fact_Cost_Live[Amount]),
    fact_Cost_Live[Type] = "Actual",
    fact_Cost_Live[Category] = "Subcontractor",
    fact_Cost_Live[Account Code] = "60201")

Subcontractor Forecast =
CALCULATE(
    SUM(fact_Cost_Live[Amount]),
    fact_Cost_Live[Type] = "Forecast",
    fact_Cost_Live[Category] = "Subcontractor")

*System Date = "DATE: " & FORMAT(TODAY(), "dd-MM-yyyy")

Total Costs FY = [Staff Cost Actual]
+[Staff Cost Forecast]
+[Subcontractor Actual]
+[Subcontractor Forecast]

*Total Costs YTD =
CALCULATE(
    [Staff Cost Actual] + [Subcontractor Actual],
    DATESYTD(dim_Date_Live[Date]))

Total Revenue YTD + Forecast = [Total Revenue YTD] + [Revenue Forecast Remaining]

Total Revenue FY = [Total Revenue YTD + Forecast]

Total Revenue YTD = [Revenue Act In-Contract] + [Revenue Act Oo-Contract]

*Total Staff Cost = 
    SUMX(
    fact_Timesheet_Live,
    fact_Timesheet_Live[Duration] *
    RELATED(dim_StaffCosts_Live[StaffCostPerHour]))
