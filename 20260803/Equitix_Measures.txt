Margin FY % = DIVIDE([Profit FY], [Total Revenue FY])

Margin YTD % = DIVIDE([Profit YTD], [Total Revenue YTD])

Profit FY = [Total Revenue FY] - [Total Costs FY]

Profit YTD = [Total Revenue YTD] - [Total Costs YTD]

Revenue = SUM(fact_Revenue_Live[Amount]) * -1

Reporting Year =
SELECTCOLUMNS(
    FILTER(
        VALUES(dim_Date_Live[Year]),
        dim_Date_Live[Year] IN { 2025, 2026 }
    ),
    "Year", dim_Date_Live[Year])

_Selected Reporting Year =
COALESCE(
    SELECTEDVALUE('Reporting Year'[Year]),
    YEAR(TODAY()))

_Reporting YTD End Date =
VAR SelectedYear = [_Selected Reporting Year]
VAR CutoverDate =
    EOMONTH(
        TODAY(),
        IF(DAY(TODAY()) >= 12, -1, -2))
RETURN
DATE(
    SelectedYear,
    MONTH(CutoverDate),
    DAY(CutoverDate))

_Revenue Act In-Contract Raw =
CALCULATE(
    -SUM(fact_Revenue_Live[Amount]),
    fact_Revenue_Live[Type] = "Actual",
    fact_Revenue_Live[Contract Type] = "In Contract")

Revenue Act In-Contract =
VAR SelectedYear = [_Selected Reporting Year]
VAR YTDEndDate = [_Reporting YTD End Date]
RETURN
CALCULATE(
    [_Revenue Act In-Contract Raw],
    FILTER(
        ALL(dim_Date_Live),
        dim_Date_Live[Date] >= DATE(SelectedYear, 1, 1)
            && dim_Date_Live[Date] <= YTDEndDate))

_Revenue Act Oo-Contract Raw =
CALCULATE(
    -SUM(fact_Revenue_Live[Amount]),
    fact_Revenue_Live[Type] = "Actual",
    fact_Revenue_Live[Contract Type] = "Out of Contract",
    fact_Revenue_Live[Account Code] = "40013")

RETURN
CALCULATE(

    FILTER(
        ALL(dim_Date_Live),
        dim_Date_Live[Date] >= DATE(SelectedYear, 1, 1)
